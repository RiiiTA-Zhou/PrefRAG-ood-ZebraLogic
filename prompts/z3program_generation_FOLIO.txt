Given a problem description and a question, the task is to formulate the problem as a Z3 logic program using python API, consisting three parts: Declarations, Premises, and Options.
Declarations: Declare the variables and functions.
Premises: Write the constraints in the context as logic formulas.
Conclusion: Write the statement in the context as logic formula.
Then, define a python function to inference if the conclusion is true, false or uncertain.
Do not give further explanation, output the code directly.
------
Example #1
Context: Phoneix's music is classified under the indie pop genre. Phoenix is a band from France. French bands write songs in French or in English. Aside from indie pop, pop rock and synth-pop are two other genres of music. Phoneix has no songs in French.

Question: Based on the above information, is the following statement true, false, or uncertain? Phoenix makes pop rock music.

Options:
A) True
B) False
C) Uncertain

Z3 Program:
```python
# Declaration
bands_sort, (Phoneix,) = EnumSort('bands', ['Phoneix'])
bands = [Phoneix]
genres_sort, (indie_pop, pop_rock, synth_pop) = EnumSort('genres', ['indie_pop', 'pop_rock', 'synth_pop'])
genres = [indie_pop, pop_rock, synth_pop]

makes = Function('makes', bands_sort, genres_sort)
from_france = Function('from_france', bands_sort, BoolSort())
in_french = Function('in_french', bands_sort, BoolSort())
in_english = Function('in_english', bands_sort, BoolSort())

# Premises
pre_conditions = []
# Phoneix's music is classified under the indie pop genre.
pre_conditions.append(makes(Phoneix)==indie_pop)
# Phoenix is a band from France.
pre_conditions.append(from_france(Phoneix))
# French bands write songs in French or in English.
b = Const('b', bands_sort)
pre_conditions.append(ForAll([b], Implies(from_france(b), Or(in_french(b), in_english(b)))))
# Phoneix has no songs in French.
pre_conditions.append(Not(in_french(Phoneix)))

# Question: Based on the above information, is the following statement true, false, or uncertain? Phoenix makes pop rock music.
conclusion = makes(Phoneix) == pop_rock

def check_statement():
    solver = Solver()
    solver.add(pre_conditions)
    # Check if statement is necessarily true
    solver.push()
    solver.add(Not(conclusion))
    if solver.check() == unsat:
        return 'A) True'
    solver.pop()
    # Check if statement is necessarily false  
    solver.push()
    solver.add(conclusion)
    if solver.check() == unsat:
        return 'B) False'
    return 'C) Uncertain'

print(check_statement())
```

h = Const('h', humans_sort)

pre_conditions.append(releases(Tyga, Well_Done_3))
# Rappers are not opera singers.
h = Const('h', humans_sort)
pre_conditions.append(ForAll([h], Implies(is_job(h, rapper), Not(is_job(h, opera_singer)))))

# Question: Based on the above information, is the following statement true, false, or uncertain? Well Done 3 is a rap album.
conclusion = is_rap_album(Well_Done_3)

    solver.push()
    solver.add(Not(conclusion))
    if solver.check() == unsat:
        return 'A) True'
    solver.pop()
    # Check if statement is necessarily false  
    solver.push()
    solver.add(conclusion)
    if solver.check() == unsat:
        return 'B) False'
    return 'C) Uncertain'

print(check_statement())

Example #3
Context: Either present their work at the conference or provide a tutorial session at the conference. All who present their work at the conference will attend in person. All those providing a tutorial session at the conference are invited to join the club. All who attend the conference in person are provided with souvenirs. All invited to join the club are provided with delicious meals. All provided with delicious meals are happy to communicate with each other during the dinner. All provided with delicious meals are invited to take a photo with the audience. James does not attend the conference in person and is not provided with souvenirs.

Question: Based on the above information, is the following statement true, false, or uncertain? James is provided with souvenirs.

Options:
A) True
B) False
C) Uncertain

Z3 Program:
```python
# Declaration
persons_sort, (James,) = EnumSort('persons', ['James'])
persons = [James]

Present = Function('Present', persons_sort, BoolSort())
Tutorial = Function('Tutorial', persons_sort, BoolSort())
InPerson = Function('InPerson', persons_sort, BoolSort())
Invited = Function('Invited', persons_sort, BoolSort())
Souvenirs = Function('Souvenirs', persons_sort, BoolSort())
Meals = Function('Meals', persons_sort, BoolSort())
HappyCommunicate = Function('HappyCommunicate', persons_sort, BoolSort())
InvitedTakePhoto = Function('InvitedTakePhoto', persons_sort, BoolSort())

# Premises
pre_conditions = []
# Either present their work at the conference or provide a tutorial session at the conference.
p = Const('p', persons_sort)
pre_conditions.append(ForAll([p], Or(Present(p), Tutorial(p))))

# All who present their work at the conference will attend in person.
pre_conditions.append(ForAll([p], Implies(Present(p), InPerson(p))))

# All those providing a tutorial session at the conference are invited to join the club.
pre_conditions.append(ForAll([p], Implies(Tutorial(p), Invited(p))))

# All who attend the conference in person are provided with souvenirs.
pre_conditions.append(ForAll([p], Implies(InPerson(p), Souvenirs(p))))

# All invited to join the club are provided with delicious meals.
pre_conditions.append(ForAll([p], Implies(Invited(p), Meals(p))))

# All provided with delicious meals are happy to communicate with each other during the dinner.
pre_conditions.append(ForAll([p], Implies(Meals(p), HappyCommunicate(p))))

# All provided with delicious meals are invited to take a photo with the audience.
pre_conditions.append(ForAll([p], Implies(Meals(p), InvitedTakePhoto(p))))

# James does not attend the conference in person and is not provided with souvenirs.
pre_conditions.append(And(Not(InPerson(James)), Not(Souvenirs(James))))

# Question: Based on the above information, is the following statement true, false, or uncertain? James is provided with souvenirs.
conclusion = Souvenirs(James)

def check_statement():
    solver = Solver()
    solver.add(pre_conditions)
    # Check if statement is necessarily true
    solver.push()
    solver.add(Not(conclusion))
    if solver.check() == unsat:
        return 'A) True'
    solver.pop()
    # Check if statement is necessarily false  
    solver.push()
    solver.add(conclusion)
    if solver.check() == unsat:
        return 'B) False'
    return 'C) Uncertain'

print(check_statement())
```
